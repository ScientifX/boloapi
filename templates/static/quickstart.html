QuickStart Guide

Get started with the BoloDoc API in minutes. This guide walks you through authentication and making your first API calls.

Step 1: Get Your API Key

Your API key is required to authenticate all API requests. You received it via email when you registered.

How to Access Your API Key:

  • Check your registration email - Your API key was sent when you signed up
  • View or reset it - Go to your Profile page and use the "Reset API Key" button
Security Warning: Never commit your API key to version control or share it publicly. Store it securely in environment variables or a key vault. If compromised, reset it immediately from your profile page.
Step 2: Get an Access Token

Exchange your API key for a JWT access token. Tokens expire after {{ token_expiry_minutes }} minutes.

Request:

POST {{ base_url }}/v1/auth/token
Content-Type: application/json

{
    "api_key": "your-api-key-here"
}

Code Examples:

// Using fetch API
const API_KEY = process.env.BOLO_API_KEY; // Store securely
const BASE_URL = "{{ base_url }}";

async function getAccessToken() {
    const response = await fetch(`${BASE_URL}/v1/auth/token`, {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify({
            api_key: API_KEY
        })
    });
    
    if (response.ok) {
        const data = await response.json();
        console.log('Access token:', data.access_token);
        console.log('Expires in:', data.expires_in, 'seconds');
        console.log('Your role:', data.role);
        return data.access_token;
    } else {
        const error = await response.text();
        throw new Error(`Failed to get token: ${response.status} - ${error}`);
    }
}

// Usage
getAccessToken()
    .then(token => console.log('Ready to make API calls!'))
    .catch(error => console.error(error));

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import com.google.gson.Gson;
import com.google.gson.JsonObject;

public class BoloAuth {
    private static final String API_KEY = System.getenv("BOLO_API_KEY");
    private static final String BASE_URL = "{{ base_url }}";
    private static final HttpClient client = HttpClient.newHttpClient();
    private static final Gson gson = new Gson();
    
    public static String getAccessToken() throws Exception {
        JsonObject requestBody = new JsonObject();
        requestBody.addProperty("api_key", API_KEY);
        
        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create(BASE_URL + "/v1/auth/token"))
            .header("Content-Type", "application/json")
            .POST(HttpRequest.BodyPublishers.ofString(gson.toJson(requestBody)))
            .build();
        
        HttpResponse response = client.send(request, 
            HttpResponse.BodyHandlers.ofString());
        
        if (response.statusCode() == 200) {
            JsonObject data = gson.fromJson(response.body(), JsonObject.class);
            String accessToken = data.get("access_token").getAsString();
            int expiresIn = data.get("expires_in").getAsInt();
            String role = data.get("role").getAsString();
            
            System.out.println("Access token: " + accessToken);
            System.out.println("Expires in: " + expiresIn + " seconds");
            System.out.println("Your role: " + role);
            
            return accessToken;
        } else {
            throw new Exception("Failed to get token: " + 
                response.statusCode() + " - " + response.body());
        }
    }
    
    public static void main(String[] args) {
        try {
            String token = getAccessToken();
            System.out.println("Ready to make API calls!");
        } catch (Exception e) {
            System.err.println("Error: " + e.getMessage());
        }
    }
}

import (
    "bytes"
    "encoding/json"
    "fmt"
    "io/ioutil"
    "net/http"
    "os"
)

const baseURL = "{{ base_url }}"

type TokenRequest struct {
    APIKey string `json:"api_key"`
}

type TokenResponse struct {
    AccessToken string `json:"access_token"`
    TokenType   string `json:"token_type"`
    ExpiresIn   int    `json:"expires_in"`
    Role        string `json:"role"`
}

func getAccessToken() (string, error) {
    apiKey := os.Getenv("BOLO_API_KEY")
    
    requestBody := TokenRequest{APIKey: apiKey}
    jsonData, err := json.Marshal(requestBody)
    if err != nil {
        return "", err
    }
    
    resp, err := http.Post(
        baseURL+"/v1/auth/token",
        "application/json",
        bytes.NewBuffer(jsonData),
    )
    if err != nil {
        return "", err
    }
    defer resp.Body.Close()
    
    body, err := ioutil.ReadAll(resp.Body)
    if err != nil {
        return "", err
    }
    
    if resp.StatusCode == 200 {
        var tokenResp TokenResponse
        err = json.Unmarshal(body, &tokenResp)
        if err != nil {
            return "", err
        }
        
        fmt.Printf("Access token: %s\n", tokenResp.AccessToken)
        fmt.Printf("Expires in: %d seconds\n", tokenResp.ExpiresIn)
        fmt.Printf("Your role: %s\n", tokenResp.Role)
        
        return tokenResp.AccessToken, nil
    }
    
    return "", fmt.Errorf("failed to get token: %d - %s", resp.StatusCode, string(body))
}

func main() {
    token, err := getAccessToken()
    if err != nil {
        fmt.Printf("Error: %v\n", err)
        return
    }
    fmt.Println("Ready to make API calls!")
}

import (
    "bytes"
    "encoding/json"
    "fmt"
    "io/ioutil"
    "net/http"
)

type Filter struct {
    Field string `json:"field"`
    Value string `json:"value"`
}

type SearchRequest struct {
    Filters []Filter `json:"filters"`
    Logic   string   `json:"logic"`
    Limit   int      `json:"limit"`
}

type SearchResult struct {
    Count   int                      `json:"count"`
    Results []map[string]interface{} `json:"results"`
}

func simpleSearch(accessToken string) (*SearchResult, error) {
    searchReq := SearchRequest{
        Filters: []Filter{
            {Field: "title", Value: "Murder*"},
        },
        Logic: "AND",
        Limit: 10,
    }
    
    jsonData, err := json.Marshal(searchReq)
    if err != nil {
        return nil, err
    }
    
    req, err := http.NewRequest(
        "POST",
        baseURL+"/v1/search/simple",
        bytes.NewBuffer(jsonData),
    )
    if err != nil {
        return nil, err
    }
    
    req.Header.Set("Authorization", "Bearer "+accessToken)
    req.Header.Set("Content-Type", "application/json")
    
    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    body, err := ioutil.ReadAll(resp.Body)
    if err != nil {
        return nil, err
    }
    
    if resp.StatusCode == 200 {
        var results SearchResult
        err = json.Unmarshal(body, &results)
        if err != nil {
            return nil, err
        }
        
        fmt.Printf("Found %d results\n", results.Count)
        
        // Display first result
        if len(results.Results) > 0 {
            first := results.Results[0]
            data := first["data"].(map[string]interface{})
            fmt.Println("\nFirst result:")
            fmt.Printf("Title: %v\n", data["title"])
            fmt.Printf("UID: %v\n", data["uid"])
        }
        
        return &results, nil
    }
    
    return nil, fmt.Errorf("search failed: %d - %s", resp.StatusCode, string(body))
}

Available Wildcard Patterns:

  • *text* - Contains "text" anywhere
  • text* - Starts with "text"
  • *text - Ends with "text"
  • text - Exact match

Common Search Fields:

  • title - Crime titles (e.g., "Murder", "Kidnapping")
  • subjects - Person names
  • description - Full text descriptions
  • caution - Warning information
  • warning_message - Additional warnings
  • details - Detailed case information
  • remarks - Additional remarks

package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "io/ioutil"
    "net/http"
)

type Rule struct {
    Field    string      `json:"field"`
    Operator string      `json:"operator"`
    Value    interface{} `json:"value"`
}

type Group struct {
    Condition string `json:"condition"`
    Rules     []Rule `json:"rules"`
}

type AdvancedSearchRequest struct {
    Groups     []Group `json:"groups"`
    GroupLogic string  `json:"group_logic"`
    Limit      int     `json:"limit"`
}

func advancedSearch(accessToken string) (*SearchResult, error) {
    searchReq := AdvancedSearchRequest{
        Groups: []Group{
            {
                Condition: "AND",
                Rules: []Rule{
                    {Field: "sex", Operator: "equals", Value: "Male"},
                    {Field: "age_min", Operator: "between", Value: []int{25, 45}},
                    {Field: "reward_max", Operator: "gte", Value: 100000},
                    {Field: "title", Operator: "contains", Value: "Murder"},
                },
            },
        },
        GroupLogic: "AND",
        Limit:      25,
    }
    
    jsonData, err := json.Marshal(searchReq)
    if err != nil {
        return nil, err
    }
    
    req, err := http.NewRequest(
        "POST",
        baseURL+"/v1/search/advanced",
        bytes.NewBuffer(jsonData),
    )
    if err != nil {
        return nil, err
    }
    
    req.Header.Set("Authorization", "Bearer "+accessToken)
    req.Header.Set("Content-Type", "application/json")
    
    client := &http.Client{}
    resp, err := client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    body, err := ioutil.ReadAll(resp.Body)
    if err != nil {
        return nil, err
    }
    
    if resp.StatusCode == 200 {
        var results SearchResult
        err = json.Unmarshal(body, &results)
        if err != nil {
            return nil, err
        }
        
        fmt.Printf("Found %d matching fugitives\n", results.Count)
        
        for _, item := range results.Results {
            data := item["data"].(map[string]interface{})
            subjects := data["subjects"].([]interface{})
            name := "Unknown"
            if len(subjects) > 0 {
                name = subjects[0].(string)
            }
            
            fmt.Printf("\n%s\n", name)
            fmt.Printf("  Crime: %v\n", data["title"])
            fmt.Printf("  Reward: %v\n", data["reward_text"])
        }
        
        return &results, nil
    }
    
    return nil, fmt.Errorf("search failed: %d - %s", resp.StatusCode, string(body))
}

Available Operators:

  • equals - Exact match (case-insensitive for strings)
  • contains - Field contains value
  • starts_with - Field starts with value
  • ends_with - Field ends with value
  • gt - Greater than (numeric fields)
  • lt - Less than (numeric fields)
  • gte - Greater than or equal (numeric fields)
  • lte - Less than or equal (numeric fields)
  • between - Between two values (requires array: [min, max])

Numeric Fields for Comparisons:

  • age_min - Minimum age
  • age_max - Maximum age
  • reward_min - Minimum reward
  • reward_max - Maximum reward
  • height_min - Minimum height (inches)
  • height_max - Maximum height (inches)
  • weight_min - Minimum weight (pounds)
  • weight_max - Maximum weight (pounds)
